\begin{frame}{Strong Scaling}
    Measures speedup for a fixed total workload as processor count increases.

    \begin{equation*}
        \text{Speedup} = \frac{T_1}{T_N} \quad \text{\footnotesize (Where N is the number of processors)}
        \end{equation*}
